# ============================================================ GUIÓN · TABLA
def gtable(rows, w1="15%", w2="23%"):
    h = (f'<table class="gt"><thead><tr><th style="width:{w1}">Pieza</th>'
         f'<th style="width:{w2}">Idea</th><th>Guion</th></tr></thead><tbody>')
    for p, idea, hook, ver, cierre in rows:
        h += (f'<tr><td class="gd">{p}</td><td>{idea}</td>'
              f'<td><span class="gh">{hook}</span>'
              f'<span class="gx">{ver}</span>'
              f'<span class="gx"><b>Cierre:</b> {cierre}</span></td></tr>')
    return h + "</tbody></table>"
